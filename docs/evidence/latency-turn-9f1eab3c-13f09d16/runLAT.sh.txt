#!/bin/bash
# FIX's latency turn: typing-latency @ 9f1eab3c (tl9) and latency-baseline @ 13f09d16 (lbs), same base a6fe7238.
# Per tree: the profile (--build: unminified bundle, restored afterwards) then the item112a probe; on tl9 also typing-latency-scroll.mjs.
# Sequential, one grant. Each step logs to its own file; the status file says where it is.
export WS_BOX_TURN=chat1-lat-20261015
SD="C:/Users/nickh/AppData/Local/Temp/claude/c--Users-nickh-writer-studio/f1a2ee78-3ca4-459c-a8bc-9aa1df8d26b4/scratchpad"
ST="$SD/lat-status.txt"
step() { # name, dir, cmd...
  local name="$1" dir="$2"; shift 2
  echo "START $name $(date -u +%H:%M:%SZ)" >> "$ST"
  ( cd "$dir" && "$@" ) > "$SD/lat-$name.out" 2>&1
  echo "END $name exit=$? $(date -u +%H:%M:%SZ)" >> "$ST"
}
W=/c/Users/nickh/writer-studio/.claude/worktrees
step tl9-profile "$W/tl9/apps/desktop" node scripts/typing-latency-profile.mjs --build
step tl9-probe   "$W/tl9/apps/desktop" node scripts/typing-latency-item112a-probe.mjs
step tl9-scroll  "$W/tl9/apps/desktop" node scripts/run-suite.mjs --no-rebuild --only typing-latency-scroll.mjs --out "$SD/lat-tl9-scroll"
step lbs-profile "$W/lbs/apps/desktop" node scripts/typing-latency-profile.mjs --build
step lbs-probe   "$W/lbs/apps/desktop" node scripts/typing-latency-item112a-probe.mjs
echo "LAT COMPLETE" >> "$ST"
